# Tavily           — Web search fallback — via native fetch
# Pollinations.ai  — Image generation (FLUX model) — via native fetch
# Hugging Face     — Image generation fallback — via native fetch
# Wandbox          — Code execution sandbox — via native fetch
# Judge0           — Code execution (via RapidAPI) — via native fetch
# Microsoft Edge TTS — Voice/speech synthesis
msedge-tts@^2.0.7          # Microsoft Edge TTS for voice generation (on-device TTS)

# ─────────────────────────────────────────────
# 🔑 Required Environment Variables (.env)
# ─────────────────────────────────────────────
# DATABASE_URL              — Neon PostgreSQL connection string
# AUTH_SECRET               — NextAuth v5 secret key
# NEXTAUTH_SECRET           — NextAuth legacy secret (same value)
# AUTH_URL                  — Production URL (e.g. https://your-app.vercel.app)
# NEXTAUTH_URL              — Production URL (same as above)
# GOOGLE_CLIENT_ID          — Google OAuth Client ID
# GOOGLE_CLIENT_SECRET      — Google OAuth Client Secret
# GROQ_API_KEY              — Groq API key (chat + router agent)
# GEMINI_API_KEY            — Google Gemini API key (fallback AI)
# SERPAPI_API_KEY           — SerpAPI key (web search)
# TAVILY_API_KEY            — Tavily API key (web search fallback)
# HUGGINGFACE_API_KEY       — Hugging Face API key (image gen fallback)
# CLOUDINARY_URL            — Cloudinary API URL
# CLOUDINARY_API            — Cloudinary API secret
# JUDGE0_API_KEY            — Judge0 RapidAPI key (code execution)
# REDIS_URL                 — (Optional) Redis URL for persistent rate limiting
